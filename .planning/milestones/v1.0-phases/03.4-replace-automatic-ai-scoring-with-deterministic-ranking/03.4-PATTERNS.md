```ts
export function experienceMinimumRequired(description: string): number | null {
  const text = description.normalize('NFKC').replace(/[–—−]/g, '-').replace(/\u00a0/g, ' ')
  const mandatoryMinima = experienceClauses(text).flatMap(parseMandatoryExperienceMinima)
  return mandatoryMinima.length > 0 ? Math.max(...mandatoryMinima) : null
}
```

**Claim analog**

```sql
-- 0025_scoring_freshness.sql:278-297
with claimable as (
  select ...
  limit batch_size
  for update skip locked
)
update ... from claimable
set claimed_at = now(),
    claimed_input_revision = desired_input_revision,
    attempts = attempts + 1
returning ...;
```

Every `security definer` function: `set search_path = ''`, schema-qualified objects, explicit revoke/grant. Browser RPCs scope via `auth.uid()`; service claim/finalize RPCs grant only to `service_role`.

**Required RPC responsibilities**

```text
save_preferences_and_start_ranking(...)
  validate -> persist -> increment desired revision -> capture run -> seed items -> state=building

claim_deterministic_ranking_work(batch_size)
  FOR UPDATE SKIP LOCKED -> capture claimed revision

finalize_deterministic_ranking_run(run_id)
  lock state/run -> prove complete expected universe
  -> if missing, seed and return not-ready
  -> bulk-copy staged results -> switch active revision/state in one transaction

retry_deterministic_ranking_run()
  auth.uid scoped -> at most one retry for captured failed revision

get_deterministic_ranking_state()
  auth.uid scoped status/retryability
```

No exact whole-run promotion analog exists; require SQL/hosted transaction verification.
